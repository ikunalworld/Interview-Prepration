CODE TO GET MAXIMUM SUM AND MINIMUM SUM OF ARRAY OF N-1 ARRAY;

function miniMaxSum(arr: number[]): void {
    
    let SortedArray = arr.sort(function(a:any, b:any) {
        return a - b;
    });
    const MinSortedArray = SortedArray.slice(0,-1);
    const MaxSortedArray = SortedArray.slice(1);
    
    const minArraySum= MinSortedArray.reduce((accumulator, currentValue) => accumulator + currentValue, 0);
    
    const maxArraySum= MaxSortedArray.reduce((accumulator, currentValue) => accumulator + currentValue, 0);
    
    
    console.log(minArraySum + ' ' + maxArraySum);

}

CODE TO CHANGE TIME FORMAT TO 24 HR;


function timeConversion(s: string): string {
    // Write your code here
    const timePeriod = s.slice(-2);
    let hours = parseInt(s.slice(0,2));
    const minutesAndSeconds = s.slice(2, -2);
    
    if(timePeriod == "AM"){
        if(hours === 12){
            hours = 0 ;
        }
    }else if(timePeriod == "PM"){
        if(hours !== 12){
            hours = hours + 12;
        };
    }
    let changehour = String(hours);
    
    if(changehour.length ==1){
        changehour = 0+changehour; 
        
    }
    return `${changehour}${minutesAndSeconds}`;

}

CODE TO GET MEDIUM NUMBER FROM ARRAY

function findMedian(arr: number[]): number {
    // Write your code here
    let sortedArray:any = arr.sort((a,b) => a - b);
    let midlength:number = (sortedArray.length-1)/2;
    const median = sortedArray[midlength]
    return median;

}

CODE TO GET DIAGONAL NUMBER SUM OF ARRAY IN ARRAY;

function diagonalDifference(arr: number[][]): number {
    // Write your code here
    var firstDiagonal: number[] = [];
    var secondDiagonal: number[] = [];
    for(let i=0; i<arr.length; i++){
        firstDiagonal.push(arr[i][i]);
        secondDiagonal.push(arr[i][arr.length - 1 - i]);
    }
    
    const sumOfFirstD = firstDiagonal.reduce((accumulator, currentValue) => accumulator + currentValue, 0);
    const sumOfSecondD = secondDiagonal.reduce((accumulator, currentValue) => accumulator + currentValue, 0);
    
    return Math.abs(sumOfFirstD - sumOfSecondD)

}

CODE TO convert into palindrome str by removing one str ;


function palindromeIndex(s) {
    const isPalindrome = str => str === str.split('').reverse().join('');
    
    const n = s.length;
    let left = 0;
    let right = n - 1;
    
    while (left < right) {
        if (s[left] !== s[right]) {
            // Check by removing the character at the left index
            if (isPalindrome(s.slice(left + 1, right + 1))) {
                return left;
            }
            // Check by removing the character at the right index
            if (isPalindrome(s.slice(left, right))) {
                return right;
            }
            return -1; // No valid index found
        }
        left++;
        right--;
    }
    
    return -1; // The string is already a palindrome
}
